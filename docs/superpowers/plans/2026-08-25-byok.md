# BYOK (Bring Your Own Key) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Let a user paste an API key for any LLM provider LiteLLM supports (OpenAI, Groq, Gemini, NVIDIA NIM, Anthropic, ...) into a form on the OpenHire landing page and run the hiring pipeline against it, request-scoped only, never persisted.

**Architecture:** A new FastAPI backend (`api/main.py`) exposes `POST /api/run`, which builds a `LiteLLMProvider` (wraps the `litellm` library) from the pasted provider/key/model, injects it into a freshly-built (non-singleton) orchestration graph, runs the pipeline, and returns the leaderboard/reports as JSON. The existing CLI path (`main.py`, env-var `OPENAI_API_KEY`) is untouched.

**Tech Stack:** Python, FastAPI, uvicorn, litellm, langgraph (existing), pytest/pytest-asyncio, httpx (FastAPI TestClient dependency).

**Spec:** `docs/superpowers/specs/2026-08-25-byok-design.md`

## Global Constraints

- Keys are request-scoped only: used for one pipeline run, then discarded. Never written to disk, a database, logs, or browser storage (spec "Goals" / "Non-goals").
- No regression to the existing CLI (`main.py`) / env-var-based flow (spec "Goals").
- Embeddings/audio providers are out of scope — LLM only (spec "Non-goals").
- `LLMPermanentError` (bad key, unknown provider/model) must surface as HTTP 401 with a message identifying the provider, not a 500 (spec "API").
- CORS enabled on the API since the landing page is statically hosted separately (spec "API").
- API key input field must be `type="password"`, never written to `localStorage`/cookies (spec "pages/index.html").

```python
"""Shared dict-to-schema conversion helpers, used by both the CLI entry
point (main.py) and the BYOK API (api/main.py) so both build identical
JobDescription/ParsedResume/InterviewTranscript objects from the same
plain-dict input shapes."""
import json
import uuid
from datetime import datetime

from schemas.job import JobDescription, Competency
from schemas.resume import ParsedResume, WorkExperience, Education
from schemas.interview import InterviewTranscript, InterviewQuestion, InterviewAnswer


def build_job_from_dict(job_dict: dict) -> JobDescription:
    """Convert loaded job dict to JobDescription schema."""
    competencies = [
        Competency(
            name=c["name"],
            weight=c.get("weight", 1.0 / len(job_dict["competencies"])),
            importance="high",
            description=""
        )
        for c in job_dict.get("competencies", [])
    ]

    return JobDescription(
        job_id=job_dict.get("job_id", f"job_{uuid.uuid4().hex[:8]}"),
        title=job_dict.get("title", "Unknown Position"),
        description=job_dict.get("description", ""),
        experience_years=job_dict.get("experience_years", 3),
        required_skills=job_dict.get("requirements", []),
        preferred_skills=job_dict.get("preferred_skills", []),
        competencies=competencies,
        raw_text=json.dumps(job_dict),
    )


def build_resume_from_dict(resume_dict: dict) -> ParsedResume:
    """Convert loaded resume dict to ParsedResume schema."""
    current_year = datetime.now().year

    work_exp = [
        WorkExperience(
            position=w.get("position", ""),
            company=w.get("company", ""),
            start_year=current_year - int(w.get("duration_years", 0)),
            duration_months=w.get("duration_years", 0) * 12 + w.get("duration_months", 0),
            description=w.get("description", ""),
            achievements=w.get("achievements", []),
        )
        for w in resume_dict.get("work_experience", [])
    ]

    education = [
        Education(
            degree=e.get("degree", ""),
            field_of_study=e.get("field", ""),
            institution=e.get("institution", ""),
            graduation_year=e.get("graduation_year", 0),
        )
        for e in resume_dict.get("education", [])
    ]

    return ParsedResume(
        candidate_id=resume_dict.get("candidate_id", f"cand_{uuid.uuid4().hex[:8]}"),
        candidate_name=resume_dict.get("candidate_name", "Unknown"),
        email=resume_dict.get("email", ""),
        phone=resume_dict.get("phone", ""),
        education=education,
        work_experience=work_exp,
        projects=resume_dict.get("projects", []),
        skills=resume_dict.get("skills", []),
        raw_text=json.dumps(resume_dict),
        total_experience_years=resume_dict.get("total_experience_years", 0),
    )


def build_transcript_from_dict(transcript_dict: dict) -> InterviewTranscript:
    """Convert loaded transcript dict to InterviewTranscript schema."""
    exchanges = []
    for ex in transcript_dict.get("exchanges", []):
        question_data = ex.get("question", {})
        answer_data = ex.get("answer", {})

        question = InterviewQuestion(
            question_id=question_data.get("question_id", f"q_{uuid.uuid4().hex[:8]}"),
            question_text=question_data.get("question_text", ""),
            category=question_data.get("category", "general"),
            difficulty=question_data.get("difficulty", "medium"),
            competency=question_data.get("competency", "general"),
            reason="",
            expected_duration_seconds=question_data.get("duration_seconds", 60),
        )

        answer = InterviewAnswer(
            question_id=question.question_id,
            answer_text=answer_data.get("answer_text", ""),
            duration_seconds=answer_data.get("duration_seconds", 60),
        )

        exchanges.append((question, answer))

    return InterviewTranscript(
        interview_id=transcript_dict.get("interview_id", f"int_{uuid.uuid4().hex[:8]}"),
        candidate_id=transcript_dict.get("candidate_id", ""),
        job_id=transcript_dict.get("job_id", ""),
        exchanges=exchanges,
        start_time=transcript_dict.get("created_at", ""),
        is_sealed=True,
    )


def serialize(obj):
    """Recursively convert pydantic models (and containers of them) to JSON-safe values."""
    if hasattr(obj, "model_dump"):
        return serialize(obj.model_dump())
    if isinstance(obj, dict):
        return {str(k): serialize(v) for k, v in obj.items()}
    if isinstance(obj, (list, tuple)):
        return [serialize(v) for v in obj]
    return obj
```

```python
"""LiteLLM-backed LLM provider — routes to whichever provider litellm
supports (OpenAI, Groq, Gemini, NVIDIA NIM, Anthropic, and 100+ others)
using a single unified call interface, so BYOK doesn't need a hand-written
client per vendor."""
import json
from typing import Any, Dict

from providers.base import LLMProvider, LLMTransientError, LLMPermanentError

try:
    import litellm
except ImportError:
    litellm = None


class LiteLLMProvider(LLMProvider):
    """LLM provider that delegates to litellm for the BYOK path."""

    def __init__(self, provider: str, api_key: str, model: str):
        if litellm is None:
            raise ImportError("litellm package required for LiteLLMProvider. Install with: pip install litellm")
        self.provider = provider
        self.api_key = api_key
        self.model = model
        self._model_id = f"{provider}/{model}"

    def _classify(self, e: Exception) -> Exception:
        """Map a litellm exception to LLMTransientError (safe to retry) or
        LLMPermanentError (retrying can never help). litellm normalizes
        every backend's errors into its own exception hierarchy
        (litellm.exceptions), which mirrors the OpenAI SDK's shape closely
        enough to classify the same way as OpenAIProvider._classify.
        Unknown exceptions default to permanent."""
        if isinstance(e, (
            litellm.RateLimitError,
            litellm.Timeout,
            litellm.APIConnectionError,
            litellm.InternalServerError,
            litellm.ServiceUnavailableError,
        )):
            return LLMTransientError(str(e))
        if isinstance(e, litellm.APIError) and getattr(e, "status_code", None) in (408, 409, 429):
            return LLMTransientError(str(e))
        return LLMPermanentError(str(e))

    async def generate(self, prompt: str, **kwargs) -> str:
        """Generate text from a prompt via litellm."""
        try:
            response = await litellm.acompletion(
                model=self._model_id,
                api_key=self.api_key,
                messages=[{"role": "user", "content": prompt}],
                temperature=kwargs.get("temperature", 0.7),
                max_tokens=kwargs.get("max_tokens", 2048),
            )
            return response.choices[0].message.content
        except Exception as e:
            raise self._classify(e) from e

    async def generate_structured(
        self, prompt: str, schema: Dict[str, Any], **kwargs
    ) -> Dict[str, Any]:
        """Generate structured output matching a schema via litellm's
        OpenAI-compatible response_format passthrough. Same error handling
        contract as OpenAIProvider.generate_structured (see providers/llm/openai.py):
        network/rate-limit/timeout/5xx -> transient (retried by BaseAgent);
        no choices/empty content/truncated/malformed JSON -> transient (a
        one-off model hiccup); auth/bad-request -> permanent."""
        try:
            response = await litellm.acompletion(
                model=self._model_id,
                api_key=self.api_key,
                messages=[{"role": "user", "content": prompt}],
                temperature=kwargs.get("temperature", 0.7),
                max_tokens=kwargs.get("max_tokens", 2048),
                response_format={
                    "type": "json_schema",
                    "json_schema": {
                        "name": "response",
                        "schema": schema,
                        "strict": True,
                    },
                },
            )
        except Exception as e:
            raise self._classify(e) from e

        if not response.choices:
            raise LLMTransientError("LiteLLM structured response contained no choices")

        choice = response.choices[0]
        finish_reason = getattr(choice, "finish_reason", None)
        content = choice.message.content if choice.message else None

        if not content:
            raise LLMTransientError("LiteLLM structured response had empty content")

        if finish_reason == "length":
            raise LLMTransientError(
                "LiteLLM structured response was truncated before completion (finish_reason=length)"
            )

        try:
            return json.loads(content)
        except (json.JSONDecodeError, ValueError, TypeError) as e:
            raise LLMTransientError(f"LiteLLM returned malformed structured JSON: {e}") from e
```

```python
"""
LLM provider factory and implementations.
"""
from typing import Optional

from config.settings import LLM_PROVIDER, OPENAI_API_KEY, OPENAI_MODEL
from providers.base import LLMProvider
from providers.llm.mock import MockLLMProvider
from providers.llm.openai import OpenAIProvider
from providers.llm.litellm_provider import LiteLLMProvider


def get_llm_provider(
    provider: Optional[str] = None,
    api_key: Optional[str] = None,
    model: Optional[str] = None,
) -> LLMProvider:
    """Factory function to get an LLM provider.

    Two modes:
    - BYOK: pass `provider` and/or `api_key` explicitly (e.g. from the
      /api/run endpoint) to always get a LiteLLMProvider configured with
      that vendor/key/model, regardless of server-side env config.
    - Default: call with no arguments to preserve existing behavior,
      reading LLM_PROVIDER/OPENAI_API_KEY/OPENAI_MODEL from the
      environment (used by main.py and BaseAgent's default construction).
    """
    if provider is not None or api_key is not None:
        if not api_key:
            raise ValueError("api_key is required for the BYOK provider path")
        if not provider:
            raise ValueError("provider is required for the BYOK provider path")
        return LiteLLMProvider(provider=provider, api_key=api_key, model=model or OPENAI_MODEL)

    if LLM_PROVIDER == "openai":
        if not OPENAI_API_KEY:
            raise ValueError(
                "OPENAI_API_KEY environment variable is required for OpenAI provider"
            )
        return OpenAIProvider(api_key=OPENAI_API_KEY, model=OPENAI_MODEL)
    elif LLM_PROVIDER == "mock":
        return MockLLMProvider()
    else:
        raise ValueError(f"Unknown LLM provider: {LLM_PROVIDER}")


__all__ = [
    "get_llm_provider",
    "LLMProvider",
    "MockLLMProvider",
    "OpenAIProvider",
    "LiteLLMProvider",
]
```

```python
"""BYOK API — accepts a caller-supplied LLM provider/key/model, runs the
hiring evaluation pipeline against it for one request, and returns the
results. The key is used only to build a LiteLLMProvider for the duration
of this request; it is never logged, stored, or included in any response."""
import uuid
from typing import Any, Dict, List, Optional

from fastapi import FastAPI, HTTPException
from fastapi.middleware.cors import CORSMiddleware
from pydantic import BaseModel, Field

from orchestration.graph import get_pipeline, PipelineState
from providers import get_llm_provider
from providers.base import LLMPermanentError, LLMTransientError
from utils.builders import (
    build_job_from_dict,
    build_resume_from_dict,
    build_transcript_from_dict,
    serialize,
)
from utils.logging import setup_logging

logger = setup_logging(__name__)

app = FastAPI(title="OpenHire BYOK API")

# The landing page (pages/index.html) is statically hosted separately from
# this API (see deploy/vercel.json, deploy/render.yaml), so CORS must be
# enabled for the browser fetch() to succeed cross-origin.
app.add_middleware(
    CORSMiddleware,
    allow_origins=["*"],
    allow_methods=["POST"],
    allow_headers=["*"],
)


class RunRequest(BaseModel):
    provider: str = Field(..., min_length=1, description="litellm provider name, e.g. 'openai', 'groq', 'gemini', 'nvidia_nim'")
    api_key: str = Field(..., min_length=1, description="API key for the given provider; used once for this request, never stored")
    model: str = Field(..., min_length=1, description="model name for the given provider, e.g. 'gpt-4o', 'llama-3.1-70b'")
    job: Dict[str, Any]
    resumes: List[Dict[str, Any]]
    interviews: Dict[str, Dict[str, Any]] = Field(default_factory=dict)


@app.post("/api/run")
async def run_pipeline(request: RunRequest) -> Dict[str, Any]:
    llm_provider = get_llm_provider(
        provider=request.provider,
        api_key=request.api_key,
        model=request.model,
    )

    job_description = build_job_from_dict(request.job)
    parsed_resumes = [build_resume_from_dict(r) for r in request.resumes]
    interview_transcripts = {
        candidate_id: build_transcript_from_dict(transcript_dict)
        for candidate_id, transcript_dict in request.interviews.items()
    }

    run_id = f"run_{uuid.uuid4().hex[:8]}"
    state = PipelineState(
        job_description_text=job_description.description,
        candidates_resume_texts=[r.raw_text for r in parsed_resumes],
        interview_transcripts=interview_transcripts,
        job_description=None,
        parsed_resumes={},
        matching_scores={},
        shortlisted_candidates=[],
        interview_questions={},
        technical_evaluations={},
        behavioral_evaluations={},
        resume_audits={},
        integrity_evaluations={},
        bias_audits={},
        candidate_scores={},
        candidate_reports={},
        leaderboard=None,
        run_id=run_id,
        errors=[],
        audit_logs=[],
    )

    pipeline = get_pipeline(llm_provider=llm_provider)

    try:
        result = await pipeline.ainvoke(state)
    except LLMPermanentError as e:
        logger.warning(f"BYOK request failed authentication for provider {request.provider}: {e}")
        raise HTTPException(
            status_code=401,
            detail=f"authentication failed for provider {request.provider}: {e}",
        )
    except LLMTransientError as e:
        logger.warning(f"BYOK request failed after retries for provider {request.provider}: {e}")
        raise HTTPException(
            status_code=502,
            detail=f"provider {request.provider} was unavailable after retries: {e}",
        )

    return {
        "run_id": run_id,
        "leaderboard": serialize(result.get("leaderboard")),
        "candidate_reports": serialize(result.get("candidate_reports")),
        "errors": result.get("errors", []),
    }
```

OpenHire's hiring evaluation pipeline can run against your own LLM

`POST /api/run`

```json
{
  "provider": "groq",
  "api_key": "your-api-key",
  "model": "llama-3.1-70b",
  "job": { "title": "...", "description": "...", "competencies": [...] },
```

`provider` is any provider name [litellm](https://docs.litellm.ai/docs/providers)

A bad or unauthorized key returns HTTP 401 with a `detail` message naming
the provider. A provider outage after retries returns HTTP 502.
